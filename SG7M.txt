local function isTargetMob(enemy)
    if not enemy or not enemy.Name then return false end
    for _, name in ipairs(_currentTargetMons) do
        if string.find(enemy.Name, name) then
            return true
        end
    end
    return false
end

local function jumpSpace()
    local hum = player.Character and player.Character:FindFirstChildWhichIsA("Humanoid")
    if hum then
        hum.Jump = true
    end
end

local function creatTweenPart()
    local part = Instance.new("Part")
    part.Size = Vector3.new(1, 1, 1)
    part.Anchored = true
    part.CanCollide = false
    part.CFrame = game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame
    part.Transparency = 1
    part.Name = "TweenPart"
    part.Parent = workspace
    return part
end

local function tp(pos)
    isTweening = false
    character.HumanoidRootPart.CFrame = pos
end

local function tween(targetCFrame)
    local TweenPart = workspace:FindFirstChild("TweenPart") or creatTweenPart()
    TweenPart.CFrame = game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame

    if getgenv().startTween then
        getgenv().startTween:Cancel()
    end

    local distance = (targetCFrame.Position - TweenPart.Position).Magnitude

    if distance <= 50 then
        tp(targetCFrame)
        return
    end

    local TweenService = game:GetService("TweenService")

    local tweenInfo = TweenInfo.new(distance / 300, Enum.EasingStyle.Linear)

    isTweening = true

    getgenv().startTween = TweenService:Create(TweenPart, tweenInfo, {CFrame = targetCFrame})
    getgenv().startTween:Play()
    getgenv().startTween.Completed:Wait()

end

local function safeSetHumanoid(hum, prop, val)
    pcall(function()
        if typeof(hum[prop]) == "number" then
            hum[prop] = val
        end
    end)
end

-- [[ ระบบตีไว — batch-hit ทุก BasePart ของมอนในรัศมี ]]
-- กรองเฉพาะมอนใน _currentTargetMons
-- ════════════════════════════════════════════════════════════
local function getEnemiesInRange(range)
    local targets = {}
    local myHrp = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    if not myHrp then return targets end

    local folder = workspace:FindFirstChild("Enemies")
    if not folder then return targets end

    for _, enemy in pairs(folder:GetChildren()) do
        if enemy == player.Character then continue end
        -- ไม่กรองชื่อ: ตีทุกมอนในรัศมี (เพราะบินไปจุดมอนเป้าหมายอยู่แล้ว)
        local root = enemy:FindFirstChild("HumanoidRootPart")
        local hum = enemy:FindFirstChild("Humanoid")
        if root and hum and hum.Health > 0 and (root.Position - myHrp.Position).Magnitude <= range then
            table.insert(targets, enemy)
        end
    end
    return targets
end

local function doAttack()
    local char = player.Character
    if not char or not char:FindFirstChildOfClass("Tool") then return end

    local enemies = getEnemiesInRange(60)
    if #enemies == 0 then return end

    local hitList = {}
    local mainPart = nil
    local priorityParts = {"Head", "UpperTorso", "Torso", "HumanoidRootPart", "RightHand", "LeftHand", "RightArm", "LeftArm"}

    for _, enemy in ipairs(enemies) do
        pcall(function() enemy:SetAttribute("inUse", nil) end)

        local chosen = nil
        for _, name in ipairs(priorityParts) do
            chosen = enemy:FindFirstChild(name)
            if chosen then break end
        end
        if not chosen then chosen = enemy.PrimaryPart end
        if not chosen then chosen = enemy:FindFirstChildWhichIsA("BasePart") end

        if chosen then
            for _, part in ipairs(enemy:GetChildren()) do
                if part:IsA("BasePart") then
                    table.insert(hitList, {enemy, part})
                end
            end
            if not mainPart then mainPart = chosen end
        end
    end

    if not mainPart or #hitList == 0 then return end

    if not char:FindFirstChild("HasBuso") then
        pcall(function()
            ReplicatedStorage.Remotes.CommF_:InvokeServer("Buso")
        end)
    end

    pcall(function()
        RegisterAttack:FireServer(-math.huge)
        RegisterHit:FireServer(mainPart, hitList, nil, sessionId)
    end)

    for _, enemy in ipairs(enemies) do
        local part = enemy:FindFirstChild("Head")
            or enemy:FindFirstChild("HumanoidRootPart")
            or enemy.PrimaryPart
        if part then
            pcall(function()
                RegisterHit:FireServer(part, {{enemy, part}}, nil, sessionId)
            end)
        end
    end

    task.delay(0.025, function()
        pcall(function()
            RegisterHit:FireServer(true, nil, nil, sessionId)
        end)
    end)
end

-- [[ ระบบรวมมอน — วงกลมล้อมรอบ max 10 / ปล่อยทุก 8 วิ ]]
-- ดึงเฉพาะมอนใน _currentTargetMons รัศมี 350 จากตำแหน่งเป้า
-- ════════════════════════════════════════════════════════════
local _B = true

local function BringEnemy(PosMon)
    if not _B then return end
    if not PosMon then return end

    local plr = player
    local t = tick() % 8

    if t <= 3 then
        local validEnemies = {}
        for _, e in pairs(workspace.Enemies:GetChildren()) do
            if #validEnemies >= 10 then break end
            if not isTargetMob(e) then continue end

            if e:FindFirstChild("Humanoid") and e.Humanoid.Health > 0 then
                if e.PrimaryPart and (e.PrimaryPart.Position - PosMon).Magnitude <= 350 then
                    table.insert(validEnemies, e)
                end
            end
        end

        for i, mob in ipairs(validEnemies) do
            local mRoot = mob.PrimaryPart or mob:FindFirstChild("HumanoidRootPart")
            if mRoot then
                local angle = (i * ((math.pi * 2) / #validEnemies)) + (tick() * 3)
                local x = math.cos(angle) * 8
                local z = math.sin(angle) * 8

                mRoot.CFrame = CFrame.new(PosMon) * CFrame.new(x, 0, z)

                safeSetHumanoid(mob.Humanoid, "WalkSpeed", 0)
                safeSetHumanoid(mob.Humanoid, "JumpPower", 0)

                if mob.Humanoid:FindFirstChild("Animator") then
                    pcall(function() mob.Humanoid.Animator:Destroy() end)
                end
                if mob.Humanoid.Sit ~= true then
                    pcall(function() mob.Humanoid.Sit = true end)
                end
                pcall(function() plr.SimulationRadius = math.huge end)
            end
        end
    else
        for _, e in pairs(workspace.Enemies:GetChildren()) do
            if not isTargetMob(e) then continue end

            if e:FindFirstChild("Humanoid") then
                safeSetHumanoid(e.Humanoid, "WalkSpeed", 16)
                safeSetHumanoid(e.Humanoid, "JumpPower", 50)
                pcall(function() e.Humanoid.Sit = false end)
            end
        end
    end
end

local function checkItem(itemList_)
    local InventoryController = require(ReplicatedStorage.Controllers.UI.Inventory)
    local ItemReplication = require(ReplicatedStorage.Util.ItemReplication)

    local count_ = 0

    for _, item in ipairs(InventoryController:GetTiles()) do
        local itemId = item.ItemId
        local uid = item.NetworkedUID
        local count = ItemReplication.Quantity.readClient(itemId, uid) or 0

        for _, target in ipairs(itemList_) do
            if itemId == target.id then
                count_ = count
            end
        end
    end

    return count_

end

local function equipTool(typeTool)
    pcall(function()
        local backpack = game:GetService("Players").LocalPlayer.Backpack:GetChildren()
        for _, v in pairs(backpack) do
            if v:IsA("Tool") and (v.ToolTip == typeTool or v.Name == typeTool) then
                humanoid:EquipTool(v)
            end
        end
    end)
end

-- หามอนใกล้สุดทั้งแมพ (แบบ B — ไม่จำกัดระยะ)
local function getConnectEnemy(_name)
    if not hrp then return nil end

    local enemySpawns = workspace._WorldOrigin.EnemySpawns:GetChildren()
    local enemySpawned = workspace.Enemies:GetChildren()
    local enemiesList = _name
    local closestTarget = nil
    local closestDistance = math.huge

    for _, enemy in pairs(enemySpawned) do
        for _, enemyName in pairs(enemiesList) do
            if string.find(enemy.Name, enemyName) and enemy:FindFirstChild("Humanoid") and enemy.Humanoid.Health > 0 then
                local targetHrp = enemy:FindFirstChild("HumanoidRootPart")
                if targetHrp then
                    local dist = (targetHrp.Position - hrp.Position).Magnitude
                    if dist < closestDistance then
                        closestDistance = dist
                        closestTarget = enemy
                    end
                end
            end
        end
    end

    if not closestTarget then
        for _, spawnPoint in pairs(enemySpawns) do
            for _, enemyName in pairs(enemiesList) do
                if string.find(spawnPoint.Name, enemyName) then
                    local dist = (spawnPoint.CFrame.Position - hrp.Position).Magnitude
                    if dist < closestDistance then
                        closestDistance = dist
                        closestTarget = spawnPoint
                    end
                end
            end
        end
    end

    if closestTarget then
        pcall(function()
            closestTarget:SetAttribute("inUse", true)
            if closestTarget:FindFirstChild("HumanoidRootPart") then
                closestTarget.HumanoidRootPart.CanCollide = false
            end
        end)
        return closestTarget
    end

    return nil
end

local function getBlackPack(item)
    return player.Backpack:FindFirstChild(item) or character:FindFirstChild(item)
end

local beli = player:FindFirstChild("Data"):FindFirstChild("Beli")
local oldBeli, newBeli = 0, 0
local chestCollected = 0

local chestBlackList = {}

local function collectChest()
    if not character then
        return
    end
    local CollectionService = game:GetService("CollectionService")
    local Position = character:GetPivot().Position
    local Chests = CollectionService:GetTagged("_ChestTagged")
    local Distance, Nearest = math.huge, nil
    for i = 1, #Chests do
        local Chest = Chests[i]
        local Magnitude = (Chest:GetPivot().Position - Position).Magnitude
        if Chest.CanTouch and Magnitude < Distance and not chestBlackList[Chest] then
            Distance = Magnitude
            Nearest = Chest
        end
    end

    if not Nearest or not Nearest.Parent then return end

    if Nearest then
        chestBlackList[Nearest] = true
        humanoid.Sit = false
        oldBeli = beli.Value
        tween(Nearest:GetPivot())
        if humanoid.Sit then
            jumpSpace()
        end
        humanoid.Jump = true
        task.wait(0.5)
        newBeli = beli.Value
        if oldBeli ~= newBeli then chestCollected = chestCollected + 1 end
        warn("Chest Collected: ", chestCollected)
    end
end

local collectedJobIds = {}
local ServerBrowserRemote = ReplicatedStorage:WaitForChild("__ServerBrowser")

local function getJobId()

        local done = 0

        for i = 1, 100 do
            task.delay(i * 1/50, function()
                local success, data = pcall(function()
                    return ServerBrowserRemote:InvokeServer(i)
                end)

                if success and data then
                    for jobId, serverData in pairs(data) do
                        table.insert(collectedJobIds, jobId)
                    end
                end

                done = done + 1
            end)
        end

        repeat task.wait() until done >= 100

end

local function hopServer()
    if not collectedJobIds[1] then getJobId() end
    local jobId = math.random(1, #collectedJobIds)
    ServerBrowserRemote:InvokeServer("teleport", collectedJobIds[jobId])
end

local function hasFOD()
    return getBlackPack("Fist of Darkness")
end

-- function นี้ใช้หา dark flag
local function Get_Dark_Flagment()
    if game:GetService("Lighting"):GetAttribute("MAP") ~= "Sea2" then tpWorld(2) end
    _currentTargetMons = {"Darkbeard"}
    local Darkbeard = getConnectEnemy({"Darkbeard"})
    if Darkbeard then
        _farming = true
        repeat task.wait()
            if Darkbeard and Darkbeard:FindFirstChild("Humanoid") then
                tween(CFrame.new(Darkbeard.HumanoidRootPart.Position) * CFrame.new(0,30,0))
                equipTool("Melee")
            else
                break
            end
        until not Darkbeard or humanoid.Health <= 0
        _farming = false
    elseif hasFOD() then
        _farming = false
        repeat task.wait()
            equipTool("Fist of Darkness")
            tween(CFrame.new(3776.93921, 14.6768322, -3499.31567, 0.609171271, -2.72086922e-08, -0.793038666, 7.68717001e-08, 1, 2.47394496e-08, 0.793038666, -7.60327907e-08, 0.609171271))
        until workspace.Enemies:FindFirstChild("Darkbeard") or not hasFOD()
        task.wait(1)
    else
        _farming = false
        repeat task.wait()
            if humanoid.Sit then
                jumpSpace()
            end
            collectChest()
        until hasFOD() or (chestCollected >= getgenv().SGConfigs["CollectChestCount"]) or workspace.Enemies:FindFirstChild("Darkbeard")

        if (chestCollected >= getgenv().SGConfigs["CollectChestCount"]) and not hasFOD() then
            while not hasFOD() do
                task.wait(0.3)
                hopServer()
            end
        end

    end

end
